        assert "confidence" in score
        assert score["confidence"] == "高"

    def test_returns_summary_key(self):
        score = generate_readiness_score(80, ConfidenceLevel.HIGH)
        assert "summary" in score
        assert isinstance(score["summary"], str)

    def test_high_completeness_high_confidence_summary(self):
        score = generate_readiness_score(85, ConfidenceLevel.HIGH)
        assert "数据完整" in score["summary"]
        assert "置信度高" in score["summary"]

    def test_medium_completeness_medium_confidence_summary(self):
        score = generate_readiness_score(55, ConfidenceLevel.MEDIUM)
        assert "基本完整" in score["summary"]
        assert "置信度中等" in score["summary"]

    def test_low_completeness_low_confidence_summary(self):
        score = generate_readiness_score(30, ConfidenceLevel.LOW)
        assert "不完整" in score["summary"]
        assert "人工复核" in score["summary"]

    def test_confidence_value_is_chinese(self):
        for level, expected in [
            (ConfidenceLevel.HIGH, "高"),
            (ConfidenceLevel.MEDIUM, "中"),
            (ConfidenceLevel.LOW, "低"),
        ]:
            score = generate_readiness_score(50, level)
            assert score["confidence"] == expected
