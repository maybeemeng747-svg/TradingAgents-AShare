# [S-007] candidate_tier_budget
"""Candidate Tier Classification & TA Token Budget Allocation.

Responsibilities:
1. Classify candidates into A (priority deep dive) / B (observe, wait for
   evidence) / C (eliminate / deprioritize) tiers based on the composite
   score, risk flags, data completeness, and strategy resonance already
   computed by the selection_priority_gate (S-005).
2. Assign a ``ta_budget_priority`` token budget suggestion per tier.
3. Explain ``missing_evidence_for_upgrade`` so the Daily Plan can show
   "what evidence is needed to promote this candidate".
4. Enforce a hard cap on the number of A-tier candidates per day so that
   TA token consumption stays bounded.

Design constraints:
- No external LLM calls.
- No strong buy/sell words.
- Does not auto-invoke TA.
- Does not connect to real billing APIs; only outputs budget *suggestions*.
- Single weak signal alone must NOT enter A tier.
- Two or more strong evidence categories with manageable risk CAN enter A.
- High-risk / low-completeness candidates at most B or C.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Optional


_STRONG_BUY_SELL_WORDS = {
    "立即买入", "重仓买入", "立即清仓", "满仓", "梭哈",
    "强力买入", "强烈推荐", "必买", "抄底", "追涨", "杀跌",
}

TIER_A = "A"
TIER_B = "B"
TIER_C = "C"
ALL_TIERS = {TIER_A, TIER_B, TIER_C}

TIER_BUDGET_MAP = {

_SEVERITY_HIGH_RISK_FLAGS = {"INQUIRY_RISK", "FINANCIAL_QUALITY_RISK"}

def _sanitize(text: str) -> str:
    for w in _STRONG_BUY_SELL_WORDS:
        if w in text:
            text = text.replace(w, "***")
    return text


def classify_candidate_tier(
    priority_rank: str = "",
    composite_score: float = 0.0,
    positive_category_count: int = 0,
    data_completeness: float = 0.0,
    missing_evidence: Optional[list[str]] = None,
    risk_flags: Optional[list[str]] = None,
    risk_penalty: float = 0.0,
    game_balance: str = "",
    gate_passed: bool = False,
    why_not_deep_ta: str = "",
) -> TierBudgetResult:
    """Classify a single candidate into A / B / C tier.

    The classification builds on the S-005 ``priority_rank`` but adds
    stricter guardrails:

    * **Tier A** requires:
      - ``positive_category_count >= 3`` **or**
        (``positive_category_count >= 2`` **and** ``composite_score >= 50``)
      - ``data_completeness >= 0.6``
      - No high-severity risk flags
      - Fewer than 3 risk flags in total
      - ``game_balance`` is ``favorable`` or ``neutral``
      - ``gate_passed`` is True
    * **Tier C** is assigned when:
      - High-severity risk flags are present **or**
      - 3 or more risk flags **or**
      - ``data_completeness < 0.3`` **or**
      - ``game_balance`` is ``fragile`` **or**
      - ``positive_category_count < 1``
    * **Tier B** covers everything else.

    Returns:
        ``TierBudgetResult`` with tier, budget, reason, and upgrade hints.
    """
    r_flags = set(risk_flags or [])
    missing = list(missing_evidence or [])

    has_high_risk = bool(r_flags & _SEVERITY_HIGH_RISK_FLAGS)
    many_risks = len(r_flags) >= 3
    heavy_penalty = risk_penalty <= -15
    fragile = game_balance == "fragile"
    favorable_or_neutral = game_balance in {"favorable", "neutral"}

    missing_for_upgrade: list[str] = []
    tier = TIER_C
    reason = ""

    # --- Tier A conditions ---
    meets_a_signals = (
        positive_category_count >= 3
        or (positive_category_count >= 2 and composite_score >= 50)
    )
    meets_a_completeness = data_completeness >= 0.6
    meets_a_risk = not has_high_risk and not many_risks and not heavy_penalty
    meets_a_game = favorable_or_neutral
    meets_a_gate = gate_passed

    if meets_a_signals and meets_a_completeness and meets_a_risk and meets_a_game and meets_a_gate:
        tier = TIER_A
        cat_desc = f"{positive_category_count}类信号共振"
        reason = f"{cat_desc}，综合分{composite_score:.0f}，风险可控，建议优先深度分析"
    else:
        # Build upgrade hints
        if not meets_a_signals:
            if positive_category_count < 2:
                missing_for_upgrade.append(f"需至少2类正向信号(当前{positive_category_count}类)")
            else:
                missing_for_upgrade.append(f"信号数{positive_category_count}类+综合分需≥50(当前{composite_score:.0f})")
        if not meets_a_completeness:
            missing_for_upgrade.append(f"数据完整度需≥60%(当前{data_completeness:.0%})")
        if not meets_a_risk:
            if has_high_risk:
                missing_for_upgrade.append(f"高风险标签需解除({', '.join(r_flags & _SEVERITY_HIGH_RISK_FLAGS)})")
            if many_risks:
                missing_for_upgrade.append(f"风险标签过多({len(r_flags)}个，需<3)")
            if heavy_penalty:
                missing_for_upgrade.append(f"风险罚分过重({risk_penalty}，需>-15)")
        if not meets_a_game:
            missing_for_upgrade.append(f"博弈平衡需改善(当前{game_balance or '未评估'})")
        if not meets_a_gate:
            missing_for_upgrade.append("S-005门控未通过")

        # --- Tier B or C ---
        meets_b_signals = positive_category_count >= 1
        meets_b_completeness = data_completeness >= 0.3
        meets_b_risk = not has_high_risk and not many_risks and not heavy_penalty
        not_fragile = not fragile

        if (
            meets_b_signals
            and meets_b_completeness
            and meets_b_risk
            and not_fragile
            and positive_category_count >= 1
        ):
            tier = TIER_B
            reason_parts = []
            if positive_category_count < 2:
                reason_parts.append(f"信号不足({positive_category_count}类)")
            elif not meets_a_completeness:
                reason_parts.append(f"完整度不足({data_completeness:.0%})")
            elif not favorable_or_neutral:
                reason_parts.append(f"博弈不佳({game_balance})")
            elif not gate_passed:
                reason_parts.append("门控未通过")
            else:
                reason_parts.append("综合条件接近A层但未完全满足")
            if missing:
                reason_parts.append(f"缺: {', '.join(missing[:3])}")
            reason = "；".join(reason_parts) if reason_parts else "观察等待更多证据"
        else:
            tier = TIER_C
            c_reasons = []
            if has_high_risk:
                c_reasons.append(f"高风险({', '.join(r_flags & _SEVERITY_HIGH_RISK_FLAGS)})")
            if many_risks:
                c_reasons.append(f"风险标签多({len(r_flags)}个)")
            if fragile:
                c_reasons.append("博弈fragile")
            if positive_category_count < 1:
                c_reasons.append("无正向信号")
            if data_completeness < 0.3:
                c_reasons.append(f"数据严重不足({data_completeness:.0%})")
            elif data_completeness < 0.5:
                c_reasons.append(f"数据不足({data_completeness:.0%})")
            if heavy_penalty:
                c_reasons.append(f"风险罚分重({risk_penalty})")
            reason = "；".join(c_reasons) if c_reasons else "综合条件不足"

    budget = TIER_BUDGET_MAP.get(tier, 0)
    not_deep = why_not_deep_ta if tier != TIER_A else ""

    reason = _sanitize(reason)
    not_deep = _sanitize(not_deep)
    for i, hint in enumerate(missing_for_upgrade):
        missing_for_upgrade[i] = _sanitize(hint)

    return TierBudgetResult(
        tier=tier,
        ta_budget_priority=budget,
        tier_reason=reason,
        missing_evidence_for_upgrade=missing_for_upgrade,
        why_not_deep_ta=not_deep,
    )

   104	      - ``data_completeness >= 0.6``
   105	      - No high-severity risk flags
   106	      - Fewer than 3 risk flags in total
   107	      - ``game_balance`` is ``favorable`` or ``neutral``
   108	      - ``gate_passed`` is True
   109	    * **Tier C** is assigned when:
   110	      - High-severity risk flags are present **or**
   111	      - 3 or more risk flags **or**
   112	      - ``data_completeness < 0.3`` **or**
   113	      - ``game_balance`` is ``fragile`` **or**
   114	      - ``positive_category_count < 1``
   115	    * **Tier B** covers everything else.
   116	
   117	    Returns:
   118	        ``TierBudgetResult`` with tier, budget, reason, and upgrade hints.
   119	    """
   120	    r_flags = set(risk_flags or [])
   121	    missing = list(missing_evidence or [])
   122	
   123	    has_high_risk = bool(r_flags & _SEVERITY_HIGH_RISK_FLAGS)
   124	    many_risks = len(r_flags) >= 3
   125	    heavy_penalty = risk_penalty <= -15
   126	    fragile = game_balance == "fragile"
   127	    favorable_or_neutral = game_balance in {"favorable", "neutral"}
   128	
   129	    missing_for_upgrade: list[str] = []
   130	    tier = TIER_C
   131	    reason = ""
   132	
   133	    # --- Tier A conditions ---
   134	    meets_a_signals = (
   135	        positive_category_count >= 3
   136	        or (positive_category_count >= 2 and composite_score >= 50)
   137	    )
   138	    meets_a_completeness = data_completeness >= 0.6
   139	    meets_a_risk = not has_high_risk and not many_risks and not heavy_penalty
   140	    meets_a_game = favorable_or_neutral
   141	    meets_a_gate = gate_passed
   142	
   143	    if meets_a_signals and meets_a_completeness and meets_a_risk and meets_a_game and meets_a_gate:
   144	        tier = TIER_A
   145	        cat_desc = f"{positive_category_count}类信号共振"
   146	        reason = f"{cat_desc}，综合分{composite_score:.0f}，风险可控，建议优先深度分析"
   147	    else:
   148	        # Build upgrade hints
   149	        if not meets_a_signals:
   150	            if positive_category_count < 2:
   151	                missing_for_upgrade.append(f"需至少2类正向信号(当前{positive_category_count}类)")
   152	            else:
   153	                missing_for_upgrade.append(f"信号数{positive_category_count}类+综合分需≥50(当前{composite_score:.0f})")
   154	        if not meets_a_completeness:
   155	            missing_for_upgrade.append(f"数据完整度需≥60%(当前{data_completeness:.0%})")
   156	        if not meets_a_risk:
   157	            if has_high_risk:
   158	                missing_for_upgrade.append(f"高风险标签需解除({', '.join(r_flags & _SEVERITY_HIGH_RISK_FLAGS)})")
   159	            if many_risks:
   160	                missing_for_upgrade.append(f"风险标签过多({len(r_flags)}个，需<3)")
   161	            if heavy_penalty:
   162	                missing_for_upgrade.append(f"风险罚分过重({risk_penalty}，需>-15)")
   163	        if not meets_a_game:
   164	            missing_for_upgrade.append(f"博弈平衡需改善(当前{game_balance or '未评估'})")
   165	        if not meets_a_gate:
   166	            missing_for_upgrade.append("S-005门控未通过")
   167	
   168	        # --- Tier B or C ---
   169	        meets_b_signals = positive_category_count >= 1
   170	        meets_b_completeness = data_completeness >= 0.3
   171	        meets_b_risk = not has_high_risk and not many_risks and not heavy_penalty
   172	        not_fragile = not fragile
   173	
   174	        if (
   175	            meets_b_signals
   176	            and meets_b_completeness
   177	            and meets_b_risk
   178	            and not_fragile
   179	            and positive_category_count >= 1
   180	        ):
   181	            tier = TIER_B
   182	            reason_parts = []
   183	            if positive_category_count < 2:
   184	                reason_parts.append(f"信号不足({positive_category_count}类)")
   185	            elif not meets_a_completeness:
   186	                reason_parts.append(f"完整度不足({data_completeness:.0%})")
   187	            elif not favorable_or_neutral:
   188	                reason_parts.append(f"博弈不佳({game_balance})")
   189	            elif not gate_passed:
   190	                reason_parts.append("门控未通过")
   191	            else:
   192	                reason_parts.append("综合条件接近A层但未完全满足")
   193	            if missing:
   194	                reason_parts.append(f"缺: {', '.join(missing[:3])}")
   195	            reason = "；".join(reason_parts) if reason_parts else "观察等待更多证据"
   196	        else:
   197	            tier = TIER_C
   198	            c_reasons = []
   199	            if has_high_risk:
   200	                c_reasons.append(f"高风险({', '.join(r_flags & _SEVERITY_HIGH_RISK_FLAGS)})")
   201	            if many_risks:
   202	                c_reasons.append(f"风险标签多({len(r_flags)}个)")
   203	            if fragile:
   204	                c_reasons.append("博弈fragile")
   205	            if positive_category_count < 1:
   206	                c_reasons.append("无正向信号")
   207	            if data_completeness < 0.3:
   208	                c_reasons.append(f"数据严重不足({data_completeness:.0%})")
   209	            elif data_completeness < 0.5:
   210	                c_reasons.append(f"数据不足({data_completeness:.0%})")
   211	            if heavy_penalty:
   212	                c_reasons.append(f"风险罚分重({risk_penalty})")
   213	            reason = "；".join(c_reasons) if c_reasons else "综合条件不足"
   214	
   215	    budget = TIER_BUDGET_MAP.get(tier, 0)
   216	    not_deep = why_not_deep_ta if tier != TIER_A else ""
   217	
   218	    reason = _sanitize(reason)
   219	    not_deep = _sanitize(not_deep)
   220	    for i, hint in enumerate(missing_for_upgrade):
   221	        missing_for_upgrade[i] = _sanitize(hint)
   222	
   223	    return TierBudgetResult(
   224	        tier=tier,
   225	        ta_budget_priority=budget,
   226	        tier_reason=reason,
   227	        missing_evidence_for_upgrade=missing_for_upgrade,
   228	        why_not_deep_ta=not_deep,
   229	    )

   232	def allocate_tier_budget(
   233	    results: list[TierBudgetResult],
   234	    a_tier_cap: int = A_TIER_HARD_CAP,
   235	) -> TierBudgetAllocation:
   236	    """Enforce the A-tier cap and compute aggregate budget.
   237	
   238	    When more candidates qualify for tier A than ``a_tier_cap``, the
   239	    lowest-scoring A-tier candidates are demoted to B.  The caller
   240	    should pre-sort ``results`` so that index 0 is the strongest
   241	    candidate.
   242	
   243	    Args:
   244	        results: Pre-sorted list of TierBudgetResult (strongest first).
   245	        a_tier_cap: Maximum number of A-tier candidates allowed.
   246	
   247	    Returns:
   248	        TierBudgetAllocation with aggregate stats and demotion info.
   249	    """
   250	    a_indices = [i for i, r in enumerate(results) if r.tier == TIER_A]
   251	
   252	    cap_applied = False
   253	    demoted: list[str] = []
   254	
   255	    if len(a_indices) > a_tier_cap:
   256	        cap_applied = True
   257	        for idx in a_indices[a_tier_cap:]:
   258	            old = results[idx]
   259	            old.tier = TIER_B
   260	            old.ta_budget_priority = TIER_BUDGET_MAP[TIER_B]
   600	
   601	    return candidate, ""

def _action_for_candidate(candidate: Candidate) -> str:
    """Determine the action for a candidate based on its signals."""
    if candidate.need_deep_ta:
        return "NEED_DEEP_TA"

    # If score is high and trigger price is close to current price
    evidence = candidate.evidence
    for strategy, ev in evidence.items():
        if isinstance(ev, dict):
            latest = ev.get("latest_close", 0)
            trigger = candidate.trigger_price or 0
            if latest > 0 and trigger > 0:
                distance = (trigger - latest) / latest
                if distance < 0.02:  # within 2% of trigger
                    return "WAIT_TRIGGER"

    return "OBSERVE"

- [P2] Keep capped B-tier entries from requesting deep TA — /Users/maybee/TradingAgents-AShare/tradingagents/tradeflow/plan_runner.py:218-222
  When more than 5 candidates qualify for A tier, the extra entries are demoted to B here, but only `tier`, `ta_budget_priority`, and `tier_reason` are copied back. Because every original A-tier candidate has `gate_passed=True`, `_build_plan_entry()` already set `need_deep_ta=True` and `action="NEED_DEEP_TA"`, so capped B-tier rows still show up as deep-TA work despite having a B budget. This also means saved `Candidate` objects remain uncapped if `save_candidates` is enabled.

- [P2] Apply tier ordering before truncating discovery results — /Users/maybee/TradingAgents-AShare/tradingagents/tradeflow/discovery.py:208-210
  In discovery mode, `top_candidates = candidates[:top_n]` is still chosen by composite score before this new tier sort runs, so an A-tier candidate with a lower composite score can be dropped from a small `top_n` result in favor of B/C candidates. That defeats the new A→B→C prioritization for the discovery output and budget allocation; the tier sort needs to happen before the Top-N cutoff.

- [P2] Restore upsert updates for identity fields — /Users/maybee/TradingAgents-AShare/tradingagents/tradeflow/candidate_engine.py:627-628
  The conflict update clause no longer updates `name`, `source`, or `strategy_tags_json`. When the same symbol/date is rescanned from a different source or with additional strategy hits, the row is upserted but those fields stay stale from the first insert, while the rest of the candidate data is refreshed.
