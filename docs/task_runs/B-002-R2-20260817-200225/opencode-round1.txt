+def _parse_readiness_from_text(text: Any) -> dict | None:
+    """Parse the persisted [C-008] readiness block from report text.
+
+    Returns ``{"data_completeness": int, "confidence": str}`` (either key may
+    be present alone) or None when the block or its values are absent/invalid.
+    """
+    if not isinstance(text, str) or not text:
+        return None
+    marker = _C008_MARKER_PATTERN.search(text)
+    if not marker:
+        return None
+    section = text[marker.end():]
+    parsed: dict[str, Any] = {}
+    completeness = _READINESS_COMPLETENESS_PATTERN.search(section)
+    if completeness:
+        value = int(completeness.group(1))
+        if 0 <= value <= 100:
+            parsed["data_completeness"] = value
+    confidence = _READINESS_CONFIDENCE_PATTERN.search(section)
+    if confidence:
+        parsed["confidence"] = confidence.group(1)
+    return parsed or None
