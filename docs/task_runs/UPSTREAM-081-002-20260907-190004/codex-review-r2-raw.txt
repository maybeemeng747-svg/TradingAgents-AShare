    35	def terminal_replay_event(job_id: str, job: Dict[str, Any]) -> Optional[Dict[str, Any]]:
    36	    """[UPSTREAM-081-002] Decide which terminal SSE event a subscriber should
    37	    receive when it missed the live ``job.completed``/``job.failed`` event.
    38	
    39	    Pure function of (job_id, job snapshot) — no store access, no I/O — shared
    40	    by the in-memory and Redis stores so their "missed terminal → replay"
    41	    semantics cannot drift. Returns the event envelope (``event`` + ``data``)
    42	    or None when the status is not terminal; the caller stamps its own
    43	    ``timestamp`` and must not write anything back to the store.
    44	    """
    45	    status = job.get("status")
    46	    if status == "completed":
    47	        # A malformed (non-dict) result must degrade to "no fallback fields"
    48	        # instead of crashing the SSE generator mid-stream.
    49	        result = job.get("result") if isinstance(job.get("result"), dict) else {}
    50	        return {
    51	            "event": "job.completed",
    52	            "data": {
    53	                "job_id": job_id,
    54	                "decision": job.get("decision"),
    55	                "direction": job.get("direction") or result.get("direction"),
    56	                "result": job.get("result"),
    57	                "risk_items": job.get("risk_items") or [],
    58	                "key_metrics": job.get("key_metrics") or [],
    59	                "confidence": job.get("confidence") or result.get("confidence"),
    60	                "target_price": job.get("target_price") or result.get("target_price"),
    61	                "stop_loss_price": job.get("stop_loss_price") or result.get("stop_loss_price"),
    62	            },
    63	        }
    64	    if status == "failed":
    65	        return {
    66	            "event": "job.failed",
    67	            "data": {
    68	                "job_id": job_id,
    69	                "error": job.get("error") or "job failed",
    70	            },
    71	        }
    72	    return None

   251	        """Thread-safe event emitter.
   252	
   253	        Uses put_nowait directly when called from the event loop, and
   254	        call_soon_threadsafe when called from a worker thread. The queue is
   255	        bounded; oldest events are dropped on overflow so a stalled SSE
   256	        consumer cannot exhaust process memory.
   257	        """
   258	        payload: Dict[str, Any] = {
   259	            "event": event,
   260	            "data": data,
   261	            "timestamp": _utcnow_iso(),
   262	        }
   263	        q = self._ensure_queue(job_id)
   264	        loop, on_loop = self._resolve_loop()
   265	
   266	        if on_loop:
   267	            self._put_with_overflow(q, payload)
   268	        elif loop is not None and loop.is_running():
   269	            try:
   270	                loop.call_soon_threadsafe(self._put_with_overflow, q, payload)
   271	            except RuntimeError:
   272	                # Loop closed; fall through to direct put.
   273	                self._put_with_overflow(q, payload)
   274	        else:
   275	            # Best-effort fallback: just push directly. The SSE consumer
   276	            # on the loop side will pick it up on the next wait_for cycle.
   277	            self._put_with_overflow(q, payload)

   281	    ) -> AsyncIterator[Dict[str, Any]]:
   282	        """Async generator yielding events for *job_id*.
   283	
   284	        On timeout:
   285	          - If job is still running, yield a ping event.
   286	          - If job is completed or failed, replay a terminal event and terminate.
   287	        On terminal events (job.completed, job.failed), yield and terminate.
   288	        When the generator exits (terminal event or client disconnect), the
   289	        backing event queue is dropped to free memory.
   290	        """
   291	        self._capture_loop()
   292	        q = self._ensure_queue(job_id)
   293	        try:
   294	            while True:
   295	                try:
   296	                    event = await asyncio.wait_for(q.get(), timeout=poll_interval)
   297	                    yield event
   298	                    if event["event"] in ("job.completed", "job.failed"):
   299	                        break
   300	                except asyncio.TimeoutError:
   301	                    with self._lock:
   302	                        job = dict(self._jobs.get(job_id, {}))
   303	                    replay = terminal_replay_event(job_id, job)
   304	                    if replay is not None:
   305	                        replay["timestamp"] = _utcnow_iso()

   309	                        "event": "ping",
   310	                        "data": {"timestamp": _utcnow_iso()},

   312	                    }

   315	                if self._job_events.get(job_id) is q:

## 五、r1 review 修复（2026-09-08，1×P1 + 1×P2）

### P1：Redis SSE 订阅错过终态事件后不重放终态

- **修复位置**：`api/job_store_redis.py` `subscribe()` 超时分支（约 :173-191）+ 新共享纯函数 `api/job_store.py::terminal_replay_event`（:35）。内存 store `InMemoryJobStore.subscribe` 同步改为调用同一纯函数（行为等价重构，防两处漂移）。
- **语义**：Redis 订阅在轮询超时发现 `status ∈ {completed, failed}` 时，按内存 store 既有 payload 契约补发 `job.completed`（含 `job_id/decision/direction/result/risk_items/key_metrics/confidence/target_price/stop_loss_price`，顶层缺失回退 `result`，列表缺省 `[]`）或 `job.failed`（含 `job_id/error`，error 兜底 `"job failed"`），随后才终止生成器。对 job hash 只读（一次 `HGETALL`，与修复前 `get_job().get("status")` 的 I/O 相同），不向 Redis 写入任何数据，不额外 publish。非终态/快照缺失仍走 ping，事件顺序与去重语义不变。
- **事件序列对比**（`/v1/jobs/{job_id}/events`，api/main.py `_stream_job_events`，订阅端晚于终态接入）：
  - 修复前：`job.ready` → `ping`×N → 流结束（无终态事件、无 `done` marker）——SSE 消费方永远等不到终态。
  - 修复后：`job.ready` → `ping`×N → `job.completed`（或 `job.failed`）→ `done` marker，与内存 store 行为一致。
- **附带的防御性收敛**：`result` 为非 dict 的脏数据不再让 SSE 生成器中途 AttributeError，只丢弃回退字段（`direction/confidence/...` 为 None），`result` 原样透传。内存/Redis 两边经同一函数同时获得该保护；合法数据下 payload 与旧实现逐字节一致（测试断言）。

### P2：终态重放/恢复语义专项测试

- **取舍**：把「错过终态 → 补发哪个事件、payload 是什么」抽成不依赖活 Redis 的纯函数 `terminal_replay_event`（`api/job_store.py`，模块级、零 I/O），单测放 `tests/test_upstream081_timeout_recovery.py`（4 个新用例，本机 Redis 未运行也执行）：completed payload 全字段+回退规则、failed/非终态/脏 result 边界、内存 store 订阅补发与纯函数逐字段一致、Redis 序列化往返后决策与内存视图完全一致。未引入 fakeredis 等新依赖。
- **集成测试**（`tests/test_job_store_redis.py`，带既有模块级 skip 守卫）：改写 `test_subscribe_timeout_ping` 尾断言（旧断言「终态后静默结束」正是 bug 行为）；新增 `test_subscribe_replays_completed_terminal_when_live_event_missed`、`test_subscribe_replays_failed_terminal_when_live_event_missed`——不发布任何 pub/sub 事件，靠超时轮询验证补发终态恰一次、payload 从 Redis hash 还原、生成器在终态收口。本机 Redis 未运行（connection refused），如实 skip（12 skipped），Redis 可用后自动生效。
