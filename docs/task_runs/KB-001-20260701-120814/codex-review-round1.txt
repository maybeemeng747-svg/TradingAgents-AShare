#!/usr/bin/env python3
"""KB-001 — Tree Work 本地知识库只读索引与健康审计 CLI。

只读扫描 ``~/Documents/knowledge/``，统计 wiki/investment、inbox、raw、index/log
覆盖情况，识别页面类型与结构缺口，输出 Markdown 审计报告。

**只读**：绝不向知识库写文件；不调用 LLM；不访问外网；不输出原文段落。

Usage::

    python scripts/audit_local_knowledge.py
    python scripts/audit_local_knowledge.py --knowledge-root ~/Documents/knowledge
    python scripts/audit_local_knowledge.py --output docs/knowledge_reports/local_knowledge_audit-2026-07-01.md
    python scripts/audit_local_knowledge.py --json  # 输出结构化 JSON 而非 Markdown
"""

from __future__ import annotations

import argparse
import json
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from tradingagents.dataflows.local_knowledge_audit import (  # noqa: E402
    audit_local_knowledge,
    default_knowledge_root,
    render_audit_report,
    suggest_audit_output_path,
)


def main() -> int:
    parser = argparse.ArgumentParser(
        description="KB-001 Tree Work 本地知识库只读审计",
    )
    parser.add_argument(
        "--knowledge-root",
        default=None,
        help="知识库根目录。默认 $AUTO_DEV_KNOWLEDGE_ROOT 或 ~/Documents/knowledge。",
    )
    parser.add_argument(
        "--output",
        default=None,
        help="Markdown 输出路径。默认 docs/knowledge_reports/local_knowledge_audit-YYYY-MM-DD.md。",
    )
    parser.add_argument(
        "--json",
        action="store_true",
        help="输出结构化 JSON 而非 Markdown（用于下游程序消费）。",
    )
    parser.add_argument(
        "--stdout",
        action="store_true",
        help="只打印到 stdout，不写文件（与 --output 互斥）。",
    )
    args = parser.parse_args()

    knowledge_root = args.knowledge_root or default_knowledge_root()
    result = audit_local_knowledge(knowledge_root)

    if args.json:
        payload = json.dumps(result.to_dict(), ensure_ascii=False, indent=2)
    else:
        payload = render_audit_report(result)

    # 默认：未指定 --output 时打印 payload 到 stdout。
    # --output: 写文件 + stderr 摘要；--stdout 强制同时打印 payload。
    print_to_stdout = args.stdout or not args.output
    if print_to_stdout:
        print(payload)
    if args.output:
        out_dir = os.path.dirname(args.output)
        if out_dir and not os.path.isdir(out_dir):
            os.makedirs(out_dir, exist_ok=True)
        with open(args.output, "w", encoding="utf-8") as fh:
            fh.write(payload)
        print(
            f"[KB-001] audit written to {args.output} "
            f"(investment_pages={result.investment_page_count}, "
            f"inbox_items={result.inbox_item_count}, "
            f"structural_gaps={len(result.structural_gaps)})",
            file=sys.stderr,
        )

    # 退出码：审计总能跑完（即使空库），缺口是预期产物；只在根目录不存在且 0 页时返回 1。
    if result.investment_page_count == 0 and result.inbox_item_count == 0 and result.raw_md_count == 0:
        return 1
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

def _split_frontmatter(text: str) -> Tuple[Optional[str], str]:
    """把开头的 ``---\\n...\\n---`` frontmatter 与正文分离。

    返回 ``(frontmatter_text_or_None, body_text)``。只匹配文件起始的 frontmatter，
    避免误把正文中出现的 ``---``（如表格分隔符）当作边界。
    """
    if not text.startswith("---"):
        return None, text
    # 跳过第一个 ---，寻找下一个独占一行的 ---
    lines = text.splitlines()
    # lines[0] 应为 "---"
    if not lines or lines[0].strip() != "---":
        return None, text
    end_idx = None
    for idx in range(1, len(lines)):
        if lines[idx].strip() == "---":
            end_idx = idx
            break
    if end_idx is None:
        return None, text
    fm_text = "\n".join(lines[1:end_idx])
    body = "\n".join(lines[end_idx + 1 :])
    return fm_text, body

   230	def _split_frontmatter(text: str) -> Tuple[Optional[str], str]:
   231	    """把开头的 ``---\\n...\\n---`` frontmatter 与正文分离。
   232	
   233	    返回 ``(frontmatter_text_or_None, body_text)``。只匹配文件起始的 frontmatter，
   234	    避免误把正文中出现的 ``---``（如表格分隔符）当作边界。
   235	    """
   236	    if not text.startswith("---"):
   237	        return None, text
   238	    # 跳过第一个 ---，寻找下一个独占一行的 ---
   239	    lines = text.splitlines()
   240	    # lines[0] 应为 "---"
   241	    if not lines or lines[0].strip() != "---":
   242	        return None, text
   243	    end_idx = None
   244	    for idx in range(1, len(lines)):
   245	        if lines[idx].strip() == "---":
   246	            end_idx = idx
   247	            break
   248	    if end_idx is None:
   249	        return None, text
   250	    fm_text = "\n".join(lines[1:end_idx])
   251	    body = "\n".join(lines[end_idx + 1 :])
   252	    return fm_text, body

   255	def _parse_frontmatter(fm_text: Optional[str]) -> Dict[str, Any]:
   256	    """解析 frontmatter 文本为 dict。缺失或失败时返回空 dict。"""
   257	    if not fm_text:
   258	        return {}
   259	    if _HAS_YAML:
   260	        try:
   261	            loaded = _yaml.safe_load(fm_text)
   262	            if isinstance(loaded, dict):
   263	                return loaded
   264	            return {}
   265	        except Exception:
   266	            # 回退到极简解析器；不能因为 yaml 报错就放弃整篇审计。
   267	            pass
   268	    return _parse_frontmatter_minimal(fm_text)
   269	
   270	
   271	def _parse_frontmatter_minimal(fm_text: str) -> Dict[str, Any]:
   272	    """极简 YAML 解析器：只处理 ``key: value`` 与 ``key: [a, b]`` / 多行列表。
   273	
   274	    适用于本项目 frontmatter 形态；非通用 YAML 解析器。
   275	    """
   276	    result: Dict[str, Any] = {}
   277	    current_key: Optional[str] = None
   278	    for raw_line in fm_text.splitlines():
   279	        if not raw_line.strip():
   280	            continue
   281	        line = raw_line.rstrip()
   282	        # 多行列表项：  - xxx
   283	        stripped = line.strip()
   284	        if stripped.startswith("- ") and current_key is not None:
   285	            val = stripped[2:].strip()
   286	            existing = result.get(current_key)
   287	            if isinstance(existing, list):
   288	                existing.append(val)
   289	            else:
   290	                result[current_key] = [val]
   291	            continue
   292	        # key: value
   293	        if ":" in line:
   294	            key, _, value = line.partition(":")
   295	            key = key.strip()
   296	            value = value.strip()
   297	            if not key:
   298	                continue
   299	            if value == "":
   300	                # 可能是多行列表的开头
   301	                result[key] = []
   302	                current_key = key
   303	            elif value.startswith("[") and value.endswith("]"):
   304	                inner = value[1:-1].strip()
   305	                if inner:
   306	                    items = _split_list_items(inner)
   307	                    result[key] = items
   308	                else:
   309	                    result[key] = []
   310	                current_key = key
   311	            else:
   312	                result[key] = value.strip('"').strip("'")
   313	                current_key = key
   314	    return result

   316	
   317	def _split_list_items(inner: str) -> List[str]:
   318	    """切分 ``[a, b, "c, d"]`` 形式的列表；保留引号内逗号。"""
   319	    items: List[str] = []
   320	    buf: List[str] = []
   321	    in_quote: Optional[str] = None
   322	    for ch in inner:
   323	        if ch in ('"', "'"):
   324	            if in_quote == ch:
   325	                in_quote = None
   326	            elif in_quote is None:
   327	                in_quote = ch
   328	            buf.append(ch)
   329	            continue
   330	        if ch == "," and in_quote is None:
   331	            items.append("".join(buf).strip())
   332	            buf = []
   333	            continue
   334	        buf.append(ch)
   335	    if buf:
   336	        items.append("".join(buf).strip())
   337	    return [it for it in items if it]

   387	    """判断 ``valid_until`` 是否早于今天；无法解析（如 ``长期``）视为未过期。"""
   388	    if not valid_until:
   389	        return False
   390	    today = today or date.today()
   391	    text = valid_until.strip()
   392	    # 接受 "2026-08-31" / "2026/08/31" / "20260831"；拒绝 "长期"/"无"/"未定"。
   393	    cleaned = re.sub(r"[/-]", "", text)
   394	    if not re.fullmatch(r"\d{8}", cleaned):
   395	        return False
   396	    try:
   397	        vu_date = date(int(cleaned[:4]), int(cleaned[4:6]), int(cleaned[6:8]))
   398	    except ValueError:

   400	    return vu_date < today

   406	def classify_page_type(
   407	    filename: str,
   408	    title: Optional[str],
   409	    frontmatter: Dict[str, Any],
   410	    section_headers: List[str],
   411	    body_hint_has_todo: bool,
   412	) -> str:
   413	    """识别 investment 页面类型。
   414	
   415	    返回 ``score_table`` / ``summary`` / ``to_be_supplemented`` /
   416	    ``company`` / ``industry`` / ``unclassified`` 之一。
   417	    """
   418	    name = filename
   419	    title_text = (title or "").lower()
   420	    name_text = name.lower()
   421	
   422	    # 待补充优先（KB-002 显式要求）。
   423	    if body_hint_has_todo or "待补充" in name or "待补充" in (title or ""):
   424	        return "to_be_supplemented"
   425	
   426	    # 评分表：标题/文件名命中关键词，或正文有 "标的列表" + 利好度/共识度。
   427	    if any(k in name or k in (title or "") for k in _SCORE_TABLE_KEYWORDS):
   428	        return "score_table"
   429	    joined_headers = " | ".join(section_headers)
   430	    if "标的列表" in joined_headers and ("利好度" in joined_headers or "共识度" in joined_headers):
   431	        return "score_table"
   432	
   433	    # 汇总表/速查表。
   434	    if any(k in name or k in (title or "") for k in _SUMMARY_TABLE_KEYWORDS):
   435	        return "summary"
   436	
   437	    report_type = _safe_str(frontmatter.get("report_type")) or ""
   438	    symbols = frontmatter.get("symbols")
   439	    has_company_symbol = _is_nonempty_field(symbols) or bool(_STOCK_CODE_FILE_RE.search(name))
   440	
   441	    # 公司页：报告类型为公司类 或 文件名带 6 位股票代码 + 单标的。
   442	    if report_type in _COMPANY_REPORT_TYPES or any(k in report_type for k in ("公司", "财报")):
   443	        return "company"
   444	    if has_company_symbol and report_type and report_type not in _INDUSTRY_REPORT_TYPES:
   445	        return "company"
   446	
   447	    # 行业页：报告类型为行业/周报/策略/深度/综述/产业链。
   448	    if report_type in _INDUSTRY_REPORT_TYPES or any(
   449	        k in report_type for k in ("行业", "周报", "策略", "深度", "综述", "产业链")
   450	    ):
   451	        return "industry"
   452	    if any(k in name or k in (title or "") for k in ("产业链", "行业", "周报", "策略", "综述")):
   453	        return "industry"
   454	
   455	    # 文件名带股票代码但无明确类型线索，倾向公司页。
   456	    if _STOCK_CODE_FILE_RE.search(name):
   457	        return "company"
   458	
   459	    # title/name 含 "评分"/"速查" 兜底。
   460	    if "评分" in name or "评分" in (title or ""):
   461	        return "score_table"
   462	
   463	    _ = (name_text, title_text)  # 保留以便后续扩展，避免未用告警
   464	    return "unclassified"

   467	def _compute_machine_readiness(page: WikiPageAudit) -> str:
   468	    """根据缺失字段与章节计算 machine_readiness: high/medium/low。"""
   469	    if page.is_to_be_supplemented:
   470	        return "low"
   471	    score = 0
   472	    total = 0
   473	    checks = [
   474	        page.has_summary_section,
   475	        page.has_risk_section,
   476	        page.has_sources_field,
   477	        page.has_symbols_field,
   478	        page.has_themes_field,
   479	    ]
   480	    for c in checks:
   481	        total += 1
   482	        if c:
   483	            score += 1
   484	    # 缺推荐字段越少，readiness 越高。
   485	    missing_penalty = len(page.missing_recommended_fields)
   486	    ratio = score / total if total else 0.0
   487	    if ratio >= 0.8 and missing_penalty <= 2:
   488	        return "high"
   489	    if ratio >= 0.4 and missing_penalty <= 5:
   490	        return "medium"
   491	    return "low"

   229	class TestFrontmatterParse:
   230	    def test_split_no_frontmatter(self):
   231	        fm, body = _split_frontmatter("hello\nworld")
   232	        assert fm is None
   233	        assert body == "hello\nworld"
   234	
   235	    def test_split_with_frontmatter(self):
   236	        text = "---\ntitle: x\ntags: [a]\n---\n\n# Body\n"
   237	        fm, body = _split_frontmatter(text)
   238	        assert fm is not None
   239	        assert "title: x" in fm
   240	        # 正文首行保留（闭合 --- 之后的换行）。
   241	        assert body.strip().startswith("# Body")
   242	
   243	    def test_split_table_separator_not_treated_as_frontmatter(self):
   244	        # 表格分隔 --- 不能被误判为 frontmatter 结束（已经在 body 中）
   245	        text = "---\ntitle: x\n---\n\n| a | b |\n|---|---|\n| 1 | 2 |\n"
   246	        fm, body = _split_frontmatter(text)
   247	        assert fm is not None
   248	        assert "title: x" in fm
   249	        assert "|---|---|" in body
   250	
   251	    def test_parse_yaml_list_inline(self):
   252	        fm = _parse_frontmatter('tags: [a, "b,c", d]')
   253	        assert fm["tags"] == ["a", "b,c", "d"]
   254	
   255	    def test_parse_yaml_multiline_list(self):
   256	        fm_text = "sources:\n  - alpha\n  - beta\n"
   257	        fm = _parse_frontmatter(fm_text)
   258	        assert fm["sources"] == ["alpha", "beta"]
   259	
   260	    def test_parse_empty_value_treated_as_empty(self):
   261	        # PyYAML 把 ``sources:`` 解析为 None；极简解析器返回 []。
   262	        # 两种实现都必须被 _is_nonempty_field 视为"无有效值"。
   263	        fm = _parse_frontmatter("sources:\ntags: [a]")
   264	        assert not _is_nonempty_field(fm.get("sources"))
   265	        assert fm["tags"] == ["a"]
   266	
   267	    def test_parse_invalid_yaml_returns_empty_dict(self):
   268	        # 损坏 frontmatter：安全降级，不抛异常
   269	        fm = _parse_frontmatter(": : : broken")
   270	        assert isinstance(fm, dict)
   271	
   272	    def test_parse_none_returns_empty(self):
   273	        assert _parse_frontmatter(None) == {}

   467	def _compute_machine_readiness(page: WikiPageAudit) -> str:
   468	    """根据缺失字段与章节计算 machine_readiness: high/medium/low。"""
   469	    if page.is_to_be_supplemented:
   470	        return "low"
   471	    score = 0
   472	    total = 0
   473	    checks = [
   474	        page.has_summary_section,
   475	        page.has_risk_section,
   476	        page.has_sources_field,
   477	        page.has_symbols_field,
   478	        page.has_themes_field,
   479	    ]
   480	    for c in checks:
   481	        total += 1
   482	        if c:
   483	            score += 1
   484	    # 缺推荐字段越少，readiness 越高。
   485	    missing_penalty = len(page.missing_recommended_fields)
   486	    ratio = score / total if total else 0.0
   487	    if ratio >= 0.8 and missing_penalty <= 2:
   488	        return "high"
   489	    if ratio >= 0.4 and missing_penalty <= 5:
   490	        return "medium"
   491	    return "low"

   704	def _collect_index_misalignment(result: LocalKnowledgeAuditResult, index_entries: List[str]) -> None:
   705	    """对比 index.md 与 investment 文件，找出缺失引用与孤儿条目。
   706	
   707	    ``index_missing_pages`` 返回完整 rel_path（与其它 gap 列表保持一致），
   708	    ``index_orphan_entries`` 返回 index 引用名（无对应文件）。
   709	    """
   710	    if not index_entries:
   711	        return
   712	    # investment 文件名（去 .md 后缀）→ 完整 rel_path 映射
   713	    stem_to_rel: Dict[str, str] = {}
   714	    for page in result.page_audits:
   715	        stem = Path(page.rel_path).stem
   716	        # 同名 stem 取字典序最小的 rel_path，保证稳定。
   717	        if stem not in stem_to_rel or page.rel_path < stem_to_rel[stem]:
   718	            stem_to_rel[stem] = page.rel_path
   719	    file_stems = set(stem_to_rel.keys())
   720	
   721	    index_set = set()
   722	    for entry in index_entries:
   723	        # entry 可能是 "investment/foo" 或 "foo" 或 "foo.md"
   724	        name = entry.split("/")[-1]
   725	        if name.endswith(".md"):
   726	            name = name[:-3]
   727	        index_set.add(name)
   728	
   729	    missing_stems = sorted(file_stems - index_set)
   730	    result.index_missing_pages = [stem_to_rel[s] for s in missing_stems]
   731	    result.index_orphan_entries = sorted(index_set - file_stems)
