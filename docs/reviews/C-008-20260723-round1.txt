class TestEndToEnd:
    def test_full_pipeline_high_quality(self):
        """High data completeness + no issues → HIGH confidence, strong actions allowed."""
        dc = calculate_data_completeness(
            has_market_data=True,
            has_sentiment_data=True,
            has_news_data=True,
            has_fundamentals_data=True,
            has_smart_money_data=True,
            has_volume_price_data=True,
            has_user_context=True,
            has_position_data=True,
        )
        conf = assess_confidence(dc)
        score = generate_readiness_score(dc, conf)
        text = format_readiness_score(score)

        assert dc == 100
        assert conf == ConfidenceLevel.HIGH
        assert score["confidence"] == "高"
        assert "数据完整度：100%" in text
        assert "置信度：高" in text

    def test_full_pipeline_low_quality(self):
        """Low data completeness → LOW confidence, strong actions forbidden."""
        dc = calculate_data_completeness(
            has_market_data=True,
            has_sentiment_data=False,
            has_news_data=False,
            has_fundamentals_data=False,
            has_smart_money_data=False,
            has_volume_price_data=False,
            has_user_context=False,
            has_position_data=False,

        conf = assess_confidence(dc)
        score = generate_readiness_score(dc, conf)
        text = format_readiness_score(score)

        assert dc == 12
        assert conf == ConfidenceLevel.LOW
        assert score["confidence"] == "低"
        assert "12%" in text

    def test_full_pipeline_medium_quality(self):
        """Medium data completeness with contradictory signals → MEDIUM confidence."""
        dc = calculate_data_completeness(
            has_market_data=True,
            has_sentiment_data=True,
            has_news_data=True,
            has_fundamentals_data=True,
